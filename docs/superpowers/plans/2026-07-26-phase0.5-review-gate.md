> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Add the two checks `parse_sdrf` is blind to — coordinate collisions (already present) and NT/AC ontology-accession mismatches — into one combined report, and expose a `review` CLI that screens a directory of SDRFs in a single command (reproducing the manual PR-#45 review).

**Architecture:** Extend `validator.py` with a pure `accession_check(sdrf_path, labels)` (label map injected for testability) plus a lazy `load_ontology_labels()` (heavy, cache-backed, not unit-tested — mirrors how `fetch_bundle` isolates network). Add a `full_report()` combining structural + accession + optional `parse_sdrf`. Add `review.py` with a CLI over a directory. Motivated by PR #45: 9 of 10 real defects passed `parse_sdrf`.

**Tech Stack:** Python 3.11 stdlib + `pandas`/`pyarrow` (for reading the parquet ontology cache, already installed with sdrf-pipelines), `pytest`.

## Global Constraints

- Package `sdrf_harness` under `src/`; Python 3.11+.
- Pure check functions take an injected `labels: dict[str,str]` (accession→label) so unit tests need no real cache or network.
- Ontology cache dir: `~/Library/Caches/sdrf-pipelines/ontologies/*.parquet` (columns `accession`, `label`). Loader is lazy and NOT unit-tested against the real cache.
- Synonym tolerance in the accession check: a pair is a mismatch only when the NT and the cache label are not equal AND neither is a substring of the other after normalising case and `[\s\-_]+` → single space (this is exactly the rule the PR-#45 review used, which correctly treated `Mycobacterium`↔`Mycolicibacterium` and `salivary gland`↔`saliva-secreting gland` as non-defects).
- No Claude/tool attribution in commit messages.
- Tests must not hit network or the real ontology cache.

def normalise_term(s: str) -> str:
    return re.sub(r"[\s\-_]+", " ", s.strip().lower()).strip()

_PAIR = re.compile(r"NT=([^;\t]+);AC=([A-Za-z]+[:_]\d+)")

def accession_check(sdrf_path, labels: dict) -> list:
    out = []
    seen = set()
    for line in Path(sdrf_path).read_text().splitlines()[1:]:
        for m in _PAIR.finditer(line):
            nt, ac = m.group(1), m.group(2).replace("_", ":").upper()
            if (nt, ac) in seen:
                continue
            seen.add((nt, ac))
            real = labels.get(ac)
            if real is None:
                continue
            a, b = normalise_term(nt), normalise_term(real)
            if a != b and a not in b and b not in a:
                out.append((nt, ac, real))
    return out
```

@dataclass
class Report:
    passed: bool
    collisions: int = 0
    ragged: int = 0
    accession_mismatches: list = field(default_factory=list)
    cli_errors: list = field(default_factory=list)

def load_ontology_labels(cache_dir=None) -> dict:
    import pandas as pd
    cache_dir = cache_dir or os.path.expanduser("~/Library/Caches/sdrf-pipelines/ontologies/")
    labels = {}
    for f in glob.glob(os.path.join(cache_dir, "*.parquet")):
        try:
            df = pd.read_parquet(f, columns=["accession", "label"])
        except Exception:
            continue
        for a, l in zip(df["accession"], df["label"]):
            if isinstance(a, str):
                labels.setdefault(a.strip().upper().replace("_", ":"), str(l).strip())
    return labels

def full_report(sdrf_path, labels=None, run_cli=False) -> Report:
    ragged, collisions = structural_check(sdrf_path)
    mism = accession_check(sdrf_path, labels) if labels else []
    cli_errors = []
    if run_cli:
        v = validate(sdrf_path, run_cli=True)
        cli_errors = [e for e in v.errors if e.startswith("parse_sdrf")]
    passed = collisions == 0 and ragged == 0 and not mism and not cli_errors
    return Report(passed=passed, collisions=collisions, ragged=ragged,
                  accession_mismatches=mism, cli_errors=cli_errors)
```

```python
# src/sdrf_harness/review.py
from __future__ import annotations
import glob, os, sys
from .validator import full_report, load_ontology_labels

def review_dir(root, labels=None, run_cli=False) -> dict:
    root = str(root)
    files = sorted(glob.glob(os.path.join(root, "**", "*.sdrf.tsv"), recursive=True))
    passed = 0; failed = []
    for f in files:
        r = full_report(f, labels=labels, run_cli=run_cli)
        if r.passed:
            passed += 1
        else:
            failed.append((os.path.relpath(f, root), r))
    return {"total": len(files), "passed": passed, "failed": failed}

def format_report(result) -> str:
    lines = [f"{result['passed']}/{result['total']} passed, {len(result['failed'])} with defects"]
    for rel, r in result["failed"]:
        bits = []
        if r.collisions:
            bits.append(f"{r.collisions} collisions")
        if r.ragged:
            bits.append(f"{r.ragged} ragged")
        if r.accession_mismatches:
            bits.append(f"{len(r.accession_mismatches)} bad accessions "
                        f"(e.g. {r.accession_mismatches[0][0]}={r.accession_mismatches[0][1]})")
        if r.cli_errors:
            bits.append("parse_sdrf error")
        lines.append(f"  {rel}: " + "; ".join(bits))
    return "\n".join(lines)

def main(argv=None) -> int:
    argv = argv if argv is not None else sys.argv[1:]
    run_cli = "--cli" in argv
    paths = [a for a in argv if not a.startswith("-")]
    if not paths:
        print("usage: python -m sdrf_harness.review <dir> [--cli]"); return 2
    labels = load_ontology_labels()
    result = review_dir(paths[0], labels=labels, run_cli=run_cli)
    print(format_report(result))
    return 0 if not result["failed"] else 1

if __name__ == "__main__":
    raise SystemExit(main())
```
